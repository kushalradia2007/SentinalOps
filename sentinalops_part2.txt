=== src/sentinalops/serving/ledger.py ===
"""Prediction ledger (chunk 34): every prediction, its outcome once matured, and the scores.

SQLite, append-only: two insert-only tables, nothing is ever updated or deleted.

- ``predictions``: one row per ``StressPrediction`` plus the feature snapshot it was scored on.
- ``outcomes``: one row per verified tick (``OutcomeLabel``), whoever produced it.

A prediction's outcome is the outcome row with the same ``symbol``, ``ts_ms`` and
``horizon_s``. Writes are idempotent (a replayed message is ignored), so the ledger can be fed
from a topic that delivers at least once.

Each matured prediction is one of: ``hit`` (alert, stress came), ``miss`` (no alert, stress
came), ``false_alarm``, ``correct_negative``, or ``void`` (outcome unknowable; excluded from
every metric). Predictions whose outcome has not arrived yet are ``pending``.
"""

from __future__ import annotations

import json
import math
import sqlite3
from contextlib import closing
from dataclasses import asdict, dataclass
from pathlib import Path

from sentinalops.schemas import OutcomeLabel, StressPrediction

_SCHEMA = """
CREATE TABLE IF NOT EXISTS predictions (
    prediction_id TEXT PRIMARY KEY,
    symbol        TEXT    NOT NULL,
    ts_ms         INTEGER NOT NULL,
    horizon_s     INTEGER NOT NULL,
    probability   REAL    NOT NULL,
    alert         INTEGER NOT NULL,
    threshold     REAL    NOT NULL,
    model_version TEXT    NOT NULL,
    scored_at_ms  INTEGER NOT NULL,
    features      TEXT
);
CREATE INDEX IF NOT EXISTS predictions_time ON predictions (symbol, ts_ms);
CREATE TABLE IF NOT EXISTS outcomes (
    symbol         TEXT    NOT NULL,
    ts_ms          INTEGER NOT NULL,
    horizon_s      INTEGER NOT NULL,
    label          INTEGER,
    void_reason    TEXT,
    source         TEXT    NOT NULL,
    verified_at_ms INTEGER NOT NULL,
    PRIMARY KEY (symbol, ts_ms, horizon_s)
);
CREATE TRIGGER IF NOT EXISTS predictions_append_only BEFORE UPDATE ON predictions
BEGIN SELECT RAISE(ABORT, 'ledger is append-only'); END;
CREATE TRIGGER IF NOT EXISTS predictions_no_delete BEFORE DELETE ON predictions
BEGIN SELECT RAISE(ABORT, 'ledger is append-only'); END;
CREATE TRIGGER IF NOT EXISTS outcomes_append_only BEFORE UPDATE ON outcomes
BEGIN SELECT RAISE(ABORT, 'ledger is append-only'); END;
CREATE TRIGGER IF NOT EXISTS outcomes_no_delete BEFORE DELETE ON outcomes
BEGIN SELECT RAISE(ABORT, 'ledger is append-only'); END;
"""

_JOINED = """
SELECT p.prediction_id, p.symbol, p.ts_ms, p.horizon_s, p.probability, p.alert, p.threshold,
       p.model_version, p.scored_at_ms, p.features, o.label, o.void_reason, o.verified_at_ms,
       o.symbol IS NOT NULL AS matured
FROM predictions p
LEFT JOIN outcomes o ON o.symbol = p.symbol AND o.ts_ms = p.ts_ms AND o.horizon_s = p.horizon_s
"""


def classify(alert: bool, label: int | None, matured: bool) -> str:
    if not matured:
        return "pending"
    if label is None:
        return "void"
    if alert:
        return "hit" if label == 1 else "false_alarm"
    return "miss" if label == 1 else "correct_negative"


@dataclass
class Scores:
    """Confusion table and probability scores over matured, non-void predictions."""

    hit: int = 0
    miss: int = 0
    false_alarm: int = 0
    correct_negative: int = 0
    void: int = 0
    pending: int = 0
    brier: float = math.nan
    base_rate: float = math.nan

    @property
    def scored(self) -> int:
        return self.hit + self.miss + self.false_alarm + self.correct_negative

    @property
    def precision(self) -> float:
        n = self.hit + self.false_alarm
        return self.hit / n if n else math.nan

    @property
    def recall(self) -> float:
        n = self.hit + self.miss
        return self.hit / n if n else math.nan

    def to_dict(self) -> dict:
        return {**asdict(self), "scored": self.scored, "precision": self.precision, "recall": self.recall}


class Ledger:
    def __init__(self, path: str | Path) -> None:
        self.path = Path(path)
        self.path.parent.mkdir(parents=True, exist_ok=True)
        with closing(self._connect()) as db:
            # WAL: readers never block the writer, and the API and the live pipeline can both write
            # (a second writer waits up to the 30 s timeout instead of failing).
            db.execute("PRAGMA journal_mode=WAL")
            db.executescript(_SCHEMA)

    def _connect(self) -> sqlite3.Connection:
        # One short-lived connection per call: safe from FastAPI's thread pool and async consumers.
        db = sqlite3.connect(self.path, timeout=30)
        db.execute("PRAGMA synchronous=NORMAL")  # safe with WAL; a crash can lose only the last commit
        db.row_factory = sqlite3.Row
        return db

    # ---- writes (34.1, 34.3) -------------------------------------------------------------------
    def add_prediction(self, pred: StressPrediction, features: dict[str, float] | None = None) -> bool:
        """Store a prediction. Returns False if this ``prediction_id`` was already stored."""
        with closing(self._connect()) as db, db:
            cur = db.execute(
                "INSERT OR IGNORE INTO predictions VALUES (?,?,?,?,?,?,?,?,?,?)",
                (pred.prediction_id, pred.symbol, pred.ts_ms, pred.horizon_s, pred.probability,
                 int(pred.alert), pred.threshold, pred.model_version, pred.scored_at_ms,
                 json.dumps(features) if features is not None else None),
            )
            return cur.rowcount == 1

    def add_predictions(self, preds: list[StressPrediction]) -> int:
        """Store many predictions in one transaction (no feature snapshots). Returns rows added."""
        with closing(self._connect()) as db, db:
            cur = db.executemany(
                "INSERT OR IGNORE INTO predictions VALUES (?,?,?,?,?,?,?,?,?,?)",
                [(p.prediction_id, p.symbol, p.ts_ms, p.horizon_s, p.probability, int(p.alert),
                  p.threshold, p.model_version, p.scored_at_ms, None) for p in preds],
            )
            return cur.rowcount

    def add_many(
        self,
        predictions: list[tuple[StressPrediction, dict[str, float] | None]],
        outcomes: list[OutcomeLabel],
    ) -> tuple[int, int]:
        """Predictions (with feature snapshots) and outcomes in one transaction. Returns rows added."""
        with closing(self._connect()) as db, db:
            before = db.total_changes
            db.executemany(
                "INSERT OR IGNORE INTO predictions VALUES (?,?,?,?,?,?,?,?,?,?)",
                [(p.prediction_id, p.symbol, p.ts_ms, p.horizon_s, p.probability, int(p.alert),
                  p.threshold, p.model_version, p.scored_at_ms,
                  json.dumps(f) if f is not None else None) for p, f in predictions],
            )
            added_p = db.total_changes - before
            db.executemany(
                "INSERT OR IGNORE INTO outcomes VALUES (?,?,?,?,?,?,?)",
                [(o.symbol, o.ts_ms, o.horizon_s, o.label, o.void_reason, o.source, o.verified_at_ms)
                 for o in outcomes],
            )
            return added_p, db.total_changes - before - added_p

    def add_outcome(self, outcome: OutcomeLabel) -> bool:
        """Store a verified outcome. The first outcome for a tick wins; replays are ignored."""
        with closing(self._connect()) as db, db:
            cur = db.execute(
                "INSERT OR IGNORE INTO outcomes VALUES (?,?,?,?,?,?,?)",
                (outcome.symbol, outcome.ts_ms, outcome.horizon_s, outcome.label, outcome.void_reason,
                 outcome.source, outcome.verified_at_ms),
            )
            return cur.rowcount == 1

    def add_outcomes(self, outcomes: list[OutcomeLabel]) -> int:
        with closing(self._connect()) as db, db:
            cur = db.executemany(
                "INSERT OR IGNORE INTO outcomes VALUES (?,?,?,?,?,?,?)",
                [(o.symbol, o.ts_ms, o.horizon_s, o.label, o.void_reason, o.source, o.verified_at_ms)
                 for o in outcomes],
            )
            return cur.rowcount

    # ---- reads (34.5, 34.6) --------------------------------------------------------------------
    def _rows(self, where: str = "", params: tuple = (), with_features: bool = True) -> list[dict]:
        query = _JOINED if with_features else _JOINED.replace("p.features,", "NULL AS features,")
        with closing(self._connect()) as db:
            rows = db.execute(f"{query} {where} ORDER BY p.ts_ms, p.prediction_id", params).fetchall()
        out = []
        for r in rows:
            d = dict(r)
            d["alert"], d["matured"] = bool(d["alert"]), bool(d["matured"])
            d["features"] = json.loads(d["features"]) if d["features"] else None
            d["result"] = classify(d["alert"], d["label"], d["matured"])
            out.append(d)
        return out

    def get(self, prediction_id: str) -> dict | None:
        rows = self._rows("WHERE p.prediction_id = ?", (prediction_id,))
        return rows[0] if rows else None

    def predictions(self, since_ms: int = 0, until_ms: int | None = None,
                    model_version: str | None = None, with_features: bool = True) -> list[dict]:
        """Joined prediction rows; ``with_features=False`` skips reading and parsing the snapshots."""
        clauses, params = ["p.ts_ms >= ?"], [since_ms]
        if until_ms is not None:
            clauses.append("p.ts_ms < ?")
            params.append(until_ms)
        if model_version is not None:
            clauses.append("p.model_version = ?")
            params.append(model_version)
        return self._rows("WHERE " + " AND ".join(clauses), tuple(params), with_features)

    def scores(self, since_ms: int = 0, until_ms: int | None = None,
               model_version: str | None = None) -> Scores:
        """Confusion table and Brier score for predictions made in ``[since_ms, until_ms)``."""
        s = Scores()
        sq_err, positives = 0.0, 0
        for r in self.predictions(since_ms, until_ms, model_version):
            setattr(s, r["result"], getattr(s, r["result"]) + 1)
            if r["result"] in ("pending", "void"):
                continue
            sq_err += (r["probability"] - r["label"]) ** 2
            positives += r["label"]
        if s.scored:
            s.brier = sq_err / s.scored
            s.base_rate = positives / s.scored
        return s

"""The promoted champion, shared by the API and the stream processor.

Reloads whenever the model file changes (the 'heal' step: promotion swaps the file) and
refuses a model whose inputs differ from the configured features.
"""

from __future__ import annotations

import threading
from pathlib import Path

import joblib
import pandas as pd


class ModelUnavailable(RuntimeError):
    """No servable model: the file is missing or expects different inputs."""


class ChampionModel:
    def __init__(self, path: str | Path, feature_order: list[str]) -> None:
        self.path = Path(path)
        self.feature_order = list(feature_order)
        self._model = None
        self._mtime = 0.0
        self._lock = threading.Lock()

    @property
    def available(self) -> bool:
        return self.path.exists()

    def current(self) -> tuple[object, str]:
        """(model, version), reloading if the file changed. Raises ``ModelUnavailable``."""
        if not self.path.exists():
            raise ModelUnavailable("model not available")
        mtime = self.path.stat().st_mtime
        with self._lock:
            if self._model is None or mtime != self._mtime:
                model = joblib.load(self.path)
                if list(getattr(model, "feature_names_in_", [])) != self.feature_order:
                    raise ModelUnavailable("model inputs do not match the configured features")
                self._model, self._mtime = model, mtime
            # The registry stamps a version on the model object, so it always matches the file read.
            return self._model, getattr(self._model, "sentinal_version_", f"champion-{int(self._mtime)}")

    def predict(self, features: dict[str, float]) -> tuple[float, str]:
        """Stress probability for one feature dict, and the model version that produced it."""
        model, version = self.current()
        row = pd.DataFrame([[features[k] for k in self.feature_order]], columns=self.feature_order)
        return float(model.predict_proba(row)[:, 1][0]), version

=== src/sentinalops/simulator/heal_sim.py ===
"""Run the heal loop against a simulated market (synthetic regimes, one tick per second).

Stands in for the live stack in tests and the demo: each step scores the step's ticks with
the served champion (and the shadow model, if any) exactly as the stream processor would,
records predictions in the ledger, records outcomes once their horizon has passed, then calls
``HealController.tick``. The result is the timeline of what the loop did.
"""

from __future__ import annotations

from dataclasses import dataclass

import pandas as pd

from sentinalops.config import DomainConfig
from sentinalops.heal.controller import HealController, TickResult
from sentinalops.heal.monitor import window_scores
from sentinalops.heal.registry import ModelRegistry
from sentinalops.schemas import OutcomeLabel, StressPrediction
from sentinalops.serving.ledger import Ledger
from sentinalops.serving.model import ChampionModel
from sentinalops.simulator.regimes import make_ticks

MINUTE_MS = 60_000


def market(segments: list[tuple[str, int]], start_ms: int, seed: int = 0) -> pd.DataFrame:
    """A training-table-shaped timeline: ``[(regime, minutes), ...]`` at one tick per second."""
    parts, t = [], start_ms
    for i, (regime, minutes) in enumerate(segments):
        n = minutes * 60
        df = make_ticks(regime, n, seed=seed + 101 * i)
        df.insert(0, "ts_ms", t + 1000 * pd.RangeIndex(n).to_numpy())
        df.insert(0, "symbol", "BTCUSDT")
        df["label_status"], df["regime"], df["source"] = "ok", regime, "synthetic"
        parts.append(df)
        t += n * 1000
    return pd.concat(parts, ignore_index=True)


@dataclass
class Step:
    now_ms: int
    regime: str
    champion: str | None
    live_ap: float
    result: TickResult


def run(
    cfg: DomainConfig,
    table: pd.DataFrame,
    registry: ModelRegistry,
    ledger: Ledger,
    controller: HealController,
    step_minutes: int = 5,
) -> list[Step]:
    names, h = cfg.feature_names, cfg.label_horizon_seconds * 1000
    champion = ChampionModel(registry.champion_path, names)
    shadow = ChampionModel(registry.shadow_path, names)
    threshold = cfg.alert_threshold
    start, end = int(table["ts_ms"].min()), int(table["ts_ms"].max()) + 1000
    matured_until = start
    steps: list[Step] = []
    for lo in range(start, end, step_minutes * MINUTE_MS):
        hi = min(lo + step_minutes * MINUTE_MS, end)
        chunk = table[(table["ts_ms"] >= lo) & (table["ts_ms"] < hi)]
        preds = []
        for served in (champion, shadow):
            if not served.available:
                continue
            model, version = served.current()
            probs = model.predict_proba(chunk[names])[:, 1]
            preds += [
                StressPrediction(symbol="BTCUSDT", ts_ms=int(ts), horizon_s=h // 1000, probability=float(p),
                                 alert=bool(p >= threshold), threshold=threshold, model_version=version)
                for ts, p in zip(chunk["ts_ms"], probs, strict=True)
            ]
        ledger.add_predictions(preds)
        # outcomes are known once the horizon has passed
        matured = table[(table["ts_ms"] >= matured_until) & (table["ts_ms"] + h < hi)]
        ledger.add_outcomes([
            OutcomeLabel(symbol="BTCUSDT", ts_ms=int(ts), horizon_s=h // 1000, label=int(y))
            for ts, y in zip(matured["ts_ms"], matured["label"], strict=True)
        ])
        matured_until = hi - h
        result = controller.tick(hi)
        state = registry.state()
        live = window_scores(ledger, hi - cfg.heal.metrics_window_minutes * MINUTE_MS, hi, state.champion)
        steps.append(Step(hi, str(chunk["regime"].iloc[-1]), state.champion, live.ap, result))
    return steps


def matured_table(table: pd.DataFrame, horizon_s: int):
    """The ``load_table`` callable for the controller: rows whose label has matured by ``now_ms``."""
    return lambda now_ms: table[table["ts_ms"] + horizon_s * 1000 < now_ms]


=== src/sentinalops/simulator/regimes.py ===
"""Synthetic feature ticks for two market regimes (chunk 3): the stand-in for recorded data.

Each row is one feature tick with the columns of ``FEATURE_ORDER`` plus ``label`` (1 if
liquidity stress starts within the horizon). The two regimes differ in *why* stress happens,
which is the shift the self-healing loop has to survive:

- **calm**: stress is preceded by the book thinning out. Top-5 depth falls over the last
  30-60 s and the spread widens; trading is quiet and balanced.
- **volatile**: stress is preceded by aggressive one-sided trading. Trade count and volume
  burst in the last 1-5 s, trade imbalance goes to one side and realized volatility jumps,
  while depth changes carry no information (they are noisy, and the book thins out often
  without stress following).

A model trained only on calm data watches the wrong features in the volatile regime. That is
the thesis test in ``tests/test_thesis.py``. The numbers are plausible for BTCUSDT but not
fitted to anything; real recorded data replaces this once chunk 12 exists.
"""

from __future__ import annotations

import numpy as np
import pandas as pd

from sentinalops.features.microstructure import FEATURE_ORDER, WINDOWS_S

REGIMES = ("calm", "volatile")
BASE_RATE = {"calm": 0.03, "volatile": 0.07}


def _window_frac(w: int, scale: float) -> float:
    """How much of a slow build-up a window of ``w`` seconds sees (0 for short, ~1 for long)."""
    return 1 - np.exp(-w / scale)


def make_ticks(regime: str = "calm", n: int = 20_000, seed: int = 0) -> pd.DataFrame:
    """``n`` shuffled feature ticks from ``regime``, with columns ``FEATURE_ORDER + ['label']``."""
    if regime not in REGIMES:
        raise ValueError(f"regime must be one of {REGIMES}, got {regime!r}")
    if n < 1:
        raise ValueError("n must be positive")
    rng = np.random.default_rng(seed)
    volatile = regime == "volatile"

    y = rng.binomial(1, BASE_RATE[regime], n)
    # Precursor strength. Markets are noisy: a fifth of stress onsets come with no warning,
    # warnings vary in strength, and a few calm-ahead ticks show a warning that fizzles out.
    warned = rng.binomial(1, 0.8, n) * rng.uniform(0.2, 1.4, n)
    false_alarm = rng.binomial(1, 0.03, n) * rng.uniform(0.2, 1.0, n)
    s = np.where(y == 1, warned, false_alarm)
    side = rng.choice([-1.0, 1.0], n)  # direction of a one-sided burst

    # --- book precursors (calm regime) ------------------------------------------------
    if volatile:
        # The book thins out often and for no reason; stress does not depend on it.
        thinning = rng.binomial(1, 0.25, n) * rng.uniform(0.1, 0.5, n)
    else:
        thinning = 0.4 * s
    depth_noise = 0.25 if volatile else 0.08

    depth_ref = rng.lognormal(np.log(12.0), 0.35, n)  # BTC on the top five levels
    depth_top5 = depth_ref * np.clip(1 - thinning + rng.normal(0, depth_noise / 2, n), 0.05, None)
    book_imb = np.clip(rng.normal(0, 0.25, n) + (0.4 * side * s if volatile else 0), -0.99, 0.99)
    bid_depth = depth_top5 * (1 + book_imb) / 2
    ask_depth = depth_top5 - bid_depth
    l1_imb = np.clip(book_imb + rng.normal(0, 0.3, n), -1, 1)

    spread_ref = 1.2e-7 * (3.0 if volatile else 1.0)
    spread_up = (0.0 if volatile else 1.2 * s) + (0.3 * thinning if volatile else 0.0)
    rel_spread = spread_ref * rng.lognormal(0, 0.25, n) * (1 + spread_up)

    cols: dict[str, np.ndarray] = {
        "rel_spread": rel_spread,
        "bid_depth": bid_depth,
        "ask_depth": ask_depth,
        "depth_top5": depth_top5,
        "depth_imbalance": book_imb,
        "l1_imbalance": l1_imb,
    }

    # --- trade precursors (volatile regime) -------------------------------------------
    base_rate = 9.0 if volatile else 3.0  # trades per second
    burst = 3.0 * s if volatile else 0.0
    base_vol = 0.004 if volatile else 0.0015  # per-sqrt-second mid volatility (log units)
    for w in WINDOWS_S:
        recent = 1 - _window_frac(w, 6.0)  # a burst dominates short windows, washes out of long ones
        build = _window_frac(w, 25.0)  # thinning shows up in longer windows
        rate = base_rate * (1 + burst * (0.3 + 0.7 * recent)) * rng.lognormal(0, 0.3, n)
        count = rng.poisson(rate * w)
        cols[f"trade_intensity_{w}s"] = count / w
        cols[f"trade_volume_{w}s"] = count * rng.lognormal(np.log(0.05), 0.5, n)
        imb = rng.normal(0, 0.3 / np.sqrt(w), n) + (0.7 * side * s * (0.4 + 0.6 * recent) if volatile else 0)
        cols[f"trade_imbalance_{w}s"] = np.where(count > 0, np.clip(imb, -1, 1), 0.0)
        vol_mult = 1 + (2.5 * s * (0.3 + 0.7 * recent) if volatile else 0)
        cols[f"realized_vol_{w}s"] = base_vol * np.sqrt(w) * vol_mult * rng.lognormal(0, 0.3, n)
        cols[f"mid_return_{w}s"] = rng.normal(0, base_vol * np.sqrt(w), n) + (
            side * s * base_vol * 3 * np.sqrt(w) if volatile else 0
        )
        cols[f"depth_change_{w}s"] = -thinning * build + rng.normal(0, depth_noise * np.sqrt(build), n)
        cols[f"spread_change_{w}s"] = rel_spread - rel_spread / (1 + spread_up * build) + rng.normal(
            0, spread_ref * 0.1, n
        )

    df = pd.DataFrame(cols)[FEATURE_ORDER]
    df["label"] = y
    return df


def example_ticks(seed: int = 0) -> tuple[dict[str, float], dict[str, float]]:
    """One clearly calm tick and one clearly pre-stress tick from the calm regime, for demos."""
    df = make_ticks("calm", 4_000, seed)
    calm = df[df["label"] == 0].iloc[0]
    stressed = df[df["label"] == 1].sort_values("depth_change_60s").iloc[0]
    return (
        {k: float(calm[k]) for k in FEATURE_ORDER},
        {k: float(stressed[k]) for k in FEATURE_ORDER},
    )

=== src/sentinalops/stream/processor.py ===
"""Stream processor (chunk 19): raw trades and depth in, feature ticks and stress predictions out.

The processor is transport-agnostic: feed it events with ``on_trade`` / ``on_depth`` (rows in
the recorder schema) and it returns what to publish. ``run`` wires it to any async event source.

Event time is ``recv_time_ms``, the same clock the offline pipeline uses.

**When a tick is emitted.** Tick ``t`` (a whole multiple of the cadence) is computed once the
watermark, the newest receive time seen on either stream, reaches ``t + allowed_lateness_ms``.
Waiting a little lets the two streams, which arrive through separate topics, catch up with
each other. ``advance(now_ms)`` moves the watermark with the wall clock, so ticks keep
coming (and come out invalid) when the feed goes silent.

**Late events (task 19.5).** An event whose receive time is at or before the last emitted
tick arrived too late to be counted in it. It is dropped and counted in ``stats.late``;
offline it would have been included, so ``late`` must stay at zero for online and offline
features to agree. Out-of-order events that are not late are inserted in order.

**Suspension (task 18.5).** ``compute_tick`` returns nothing while the book is stale and
until the feed has been continuous for the longest window again, so a stale feed yields no
feature ticks and no predictions, and recovery needs a full warm-up. No extra logic.

**Buffers (task 19.2).** Trades are kept for the longest window. Depth is kept for the longest
window plus the one snapshot just before it, which is exactly what ``compute_tick`` needs to
give the same answer as on the full history (tested; dropping that snapshot breaks the test).
A further ``stale_ms`` of depth is kept as a safety margin.
"""

from __future__ import annotations

import bisect
from collections.abc import AsyncIterator, Awaitable, Callable
from dataclasses import dataclass, field

import pandas as pd

from sentinalops.config import DomainConfig
from sentinalops.data.recorder.messages import AGG_TRADE, DEPTH5, DEPTH_SCHEMA, TRADE_SCHEMA
from sentinalops.features.microstructure import compute_tick
from sentinalops.logging import get_logger
from sentinalops.schemas import FeatureTick, StressPrediction
from sentinalops.serving.model import ChampionModel, ModelUnavailable

log = get_logger("sentinalops.stream")

FEATURES_TOPIC = "features"
PREDICTIONS_TOPIC = "predictions"
Output = tuple[str, FeatureTick | StressPrediction]


@dataclass
class ProcessorStats:
    events: dict[str, int] = field(default_factory=lambda: {AGG_TRADE: 0, DEPTH5: 0})
    late: dict[str, int] = field(default_factory=lambda: {AGG_TRADE: 0, DEPTH5: 0})
    other_symbol: int = 0
    ticks: int = 0
    feature_ticks: int = 0
    suspended_ticks: int = 0  # ticks with no valid features: stale feed or warm-up
    predictions: int = 0
    shadow_predictions: int = 0
    scoring_errors: int = 0
    last_tick_ms: int | None = None


class _Buffer:
    """Rows of one stream ordered by receive time."""

    def __init__(self) -> None:
        self.times: list[int] = []
        self.rows: list[dict] = []

    def insert(self, row: dict) -> None:
        t = row["recv_time_ms"]
        if not self.times or t >= self.times[-1]:
            self.times.append(t)
            self.rows.append(row)
        else:
            i = bisect.bisect_right(self.times, t)
            self.times.insert(i, t)
            self.rows.insert(i, row)

    def drop_through(self, cutoff_ms: int, keep_one_before: bool) -> None:
        """Forget rows with time <= cutoff (keeping the newest such row if asked)."""
        n = bisect.bisect_right(self.times, cutoff_ms)
        if keep_one_before:
            n = max(n - 1, 0)
        if n:
            del self.times[:n]
            del self.rows[:n]

    def frame(self, columns: list[str]) -> pd.DataFrame:
        return pd.DataFrame(self.rows, columns=columns)


class StreamProcessor:
    def __init__(
        self,
        cfg: DomainConfig,
        model: ChampionModel | None = None,
        allowed_lateness_ms: int = 500,
        shadow: ChampionModel | None = None,
        alert_threshold: float | None = None,
    ) -> None:
        if allowed_lateness_ms < 0:
            raise ValueError("allowed_lateness_ms must be >= 0")
        self.cfg = cfg
        self.params = cfg.feature_params()
        self.model = model
        self.shadow = shadow  # the previous champion during a shadow period (chunk 45)
        self.lateness = allowed_lateness_ms
        self.threshold = cfg.alert_threshold if alert_threshold is None else alert_threshold
        self.step = self.params.cadence_s * 1000
        self.trades, self.depth = _Buffer(), _Buffer()
        self.watermark: int | None = None
        self.next_tick: int | None = None
        self.stats = ProcessorStats()

    # ---- input -------------------------------------------------------------------------------
    def on_trade(self, row: dict) -> list[Output]:
        return self._on_event(AGG_TRADE, row)

    def on_depth(self, row: dict) -> list[Output]:
        return self._on_event(DEPTH5, row)

    def on_event(self, kind: str, row: dict) -> list[Output]:
        if kind not in (AGG_TRADE, DEPTH5):
            raise ValueError(f"unknown stream kind {kind!r}")
        return self._on_event(kind, row)

    def _on_event(self, kind: str, row: dict) -> list[Output]:
        if row.get("symbol") != self.cfg.symbol:
            self.stats.other_symbol += 1
            return []
        t = int(row["recv_time_ms"])
        self.stats.events[kind] += 1
        if self.stats.last_tick_ms is not None and t <= self.stats.last_tick_ms:
            self.stats.late[kind] += 1
            return []
        (self.trades if kind == AGG_TRADE else self.depth).insert(row)
        if self.next_tick is None:
            self.next_tick = -(-t // self.step) * self.step  # first whole tick at or after the first event
        return self.advance(t)

    # ---- clock -------------------------------------------------------------------------------
    def advance(self, now_ms: int) -> list[Output]:
        """Move the watermark to ``now_ms`` (if later) and emit every tick that is now due."""
        if self.watermark is None or now_ms > self.watermark:
            self.watermark = now_ms
        out: list[Output] = []
        while self.next_tick is not None and self.next_tick + self.lateness <= self.watermark:
            out += self._emit(self.next_tick)
            self.next_tick += self.step
        return out

    def _emit(self, ts_ms: int) -> list[Output]:
        self.stats.ticks += 1
        self.stats.last_tick_ms = ts_ms
        # Forget what this tick (and so every later one) can no longer use.
        self.trades.drop_through(ts_ms - self.params.max_window_ms, keep_one_before=False)
        self.depth.drop_through(ts_ms - self.params.buffer_ms, keep_one_before=True)

        feats = self._features(ts_ms)
        if feats is None:
            self.stats.suspended_ticks += 1
            return []
        tick = FeatureTick(symbol=self.cfg.symbol, ts_ms=ts_ms, features=feats)
        self.stats.feature_ticks += 1
        out: list[Output] = [(FEATURES_TOPIC, tick)]
        if self.model is not None:
            try:
                p, version = self.model.predict(feats)
            except ModelUnavailable as e:
                self.stats.scoring_errors += 1
                log.warning("cannot score", extra={"ts_ms": ts_ms, "error": str(e)})
                return out
            out.append((PREDICTIONS_TOPIC, self._prediction(ts_ms, p, version)))
            self.stats.predictions += 1
        if self.shadow is not None and self.shadow.available:
            try:  # shadow scores go to the ledger under their own version; alerting uses the champion's
                p, version = self.shadow.predict(feats)
                out.append((PREDICTIONS_TOPIC, self._prediction(ts_ms, p, version)))
                self.stats.shadow_predictions += 1
            except ModelUnavailable:
                pass
        return out

    def _prediction(self, ts_ms: int, p: float, version: str) -> StressPrediction:
        return StressPrediction(
            symbol=self.cfg.symbol, ts_ms=ts_ms, horizon_s=self.cfg.label_horizon_seconds,
            probability=p, alert=p >= self.threshold, threshold=self.threshold, model_version=version,
        )

    def _features(self, ts_ms: int) -> dict[str, float] | None:
        if not self.depth.rows:
            return None
        depth = self.depth.frame(DEPTH_SCHEMA.names)
        trades = self.trades.frame(TRADE_SCHEMA.names)
        tick = compute_tick(depth, trades, ts_ms, self.params)
        if tick is None:
            return None
        return {name: tick[name] for name in self.params.names}


# ---- wiring ----------------------------------------------------------------------------------------
Publish = Callable[[str, str, dict], Awaitable[None]]  # (topic, key, value)


async def run(
    processor: StreamProcessor, events: AsyncIterator[tuple[str, dict]], publish: Publish
) -> ProcessorStats:
    """Consume ``(kind, row)`` events until the source ends and publish every output keyed by symbol."""
    async for kind, row in events:
        for topic, msg in processor.on_event(kind, row):
            await publish(topic, msg.symbol, msg.model_dump(mode="json"))
    return processor.stats


=== src/sentinalops/stream/verifier.py ===
"""Outcome verifier (chunk 34): the live version of the chunk 9 label.

Fed the raw depth stream, it decides for every tick ``t`` whether liquidity stress started in
``(t, t + H]``, once tick ``t + H`` has been seen, and returns an ``OutcomeLabel``. It
re-implements ``labels.build_labels`` incrementally, step for step:

1. as-of book state per tick: newest snapshot with ``recv_time_ms <= t``; unknown if older
   than ``stale_ms`` (``labels.to_ticks``),
2. trailing reference: median of the previous ``reference_ticks`` ticks, ignoring unknown
   ticks, trusted once ``min_reference_ticks`` are known (``labels.add_stress``),
3. ``stressed`` and ``confirmed`` (stressed for ``confirm_ticks`` ticks in a row),
4. the onset label at ``t`` from ticks ``t+1 .. t+H`` (``labels.add_labels``).

The replay test checks it against ``build_labels`` and ``label_at`` tick by tick.

Void outcomes (``label=None``) carry a reason:

- ``already_stressed``: the market was stressed at ``t``; onset is not defined.
- ``gap``: the book was stale somewhere in ``[t - confirm_ticks + 1, t + H]``.
- ``warmup``: otherwise unknown, i.e. the trailing reference was not established yet.

Clock, watermark and late events follow the stream processor (``docs/streaming.md``).
"""

from __future__ import annotations

import bisect
from collections import deque
from collections.abc import AsyncIterator
from dataclasses import dataclass, field

import numpy as np

from sentinalops.config import DomainConfig
from sentinalops.data.recorder.messages import DEPTH5
from sentinalops.schemas import OutcomeLabel
from sentinalops.stream.processor import Publish

OUTCOMES_TOPIC = "outcomes"
NAN = float("nan")


@dataclass
class VerifierStats:
    snapshots: int = 0
    late: int = 0
    other_symbol: int = 0
    rejected_books: int = 0  # crossed or empty, ignored like labels.book_state does
    ticks: int = 0
    outcomes: dict[str, int] = field(
        default_factory=lambda: {"1": 0, "0": 0, "already_stressed": 0, "gap": 0, "warmup": 0}
    )
    last_tick_ms: int | None = None


@dataclass
class _Tick:
    ts_ms: int
    stale: bool
    stressed: float  # 0, 1 or NaN
    confirmed: float  # 0, 1 or NaN


class OutcomeVerifier:
    def __init__(self, cfg: DomainConfig, allowed_lateness_ms: int = 500) -> None:
        if allowed_lateness_ms < 0:
            raise ValueError("allowed_lateness_ms must be >= 0")
        self.cfg = cfg
        self.p = cfg.label_params()
        self.lateness = allowed_lateness_ms
        self.step = self.p.cadence_s * 1000
        self.min_ref = min(self.p.min_reference_ticks, self.p.reference_ticks)
        # snapshot buffer: (recv_time_ms, depth_top5, rel_spread)
        self._snap_t: list[int] = []
        self._snap_v: list[tuple[float, float]] = []
        self._ref_depth: deque[float] = deque(maxlen=self.p.reference_ticks)
        self._ref_spread: deque[float] = deque(maxlen=self.p.reference_ticks)
        self._recent_stressed: deque[float] = deque(maxlen=self.p.confirm_ticks)
        # ticks still needed: t - H (to resolve) .. now, plus confirm_ticks before for the gap reason
        self._history: deque[_Tick] = deque(maxlen=self.p.horizon_ticks + self.p.confirm_ticks + 1)
        self.watermark: int | None = None
        self.next_tick: int | None = None
        self.stats = VerifierStats()

    # ---- input -------------------------------------------------------------------------------
    def on_depth(self, row: dict) -> list[OutcomeLabel]:
        if row.get("symbol") != self.cfg.symbol:
            self.stats.other_symbol += 1
            return []
        t = int(row["recv_time_ms"])
        self.stats.snapshots += 1
        if self.stats.last_tick_ms is not None and t <= self.stats.last_tick_ms:
            self.stats.late += 1
            return []
        bid, ask = float(row["bid_px_1"]), float(row["ask_px_1"])
        total = sum(float(row[f"{s}_qty_{i}"]) for s in ("bid", "ask") for i in range(1, 6))
        if not (bid > 0 and ask > bid and total > 0):
            self.stats.rejected_books += 1
            return self.advance(t)
        mid = (bid + ask) / 2
        i = bisect.bisect_right(self._snap_t, t)
        self._snap_t.insert(i, t)
        self._snap_v.insert(i, (total, (ask - bid) / mid))
        if self.next_tick is None:
            self.next_tick = -(-t // self.step) * self.step
        return self.advance(t)

    def on_event(self, kind: str, row: dict) -> list[OutcomeLabel]:
        """Accepts the combined stream; only depth matters for the label."""
        return self.on_depth(row) if kind == DEPTH5 else []

    # ---- clock -------------------------------------------------------------------------------
    def advance(self, now_ms: int) -> list[OutcomeLabel]:
        if self.watermark is None or now_ms > self.watermark:
            self.watermark = now_ms
        out: list[OutcomeLabel] = []
        while self.next_tick is not None and self.next_tick + self.lateness <= self.watermark:
            resolved = self._tick(self.next_tick)
            if resolved is not None:
                out.append(resolved)
            self.next_tick += self.step
        return out

    # ---- one tick ----------------------------------------------------------------------------
    def _state(self, ts_ms: int) -> tuple[float, float, bool]:
        """(depth_top5, rel_spread, stale) as of ``ts_ms``; drops snapshots no later tick needs."""
        i = bisect.bisect_right(self._snap_t, ts_ms) - 1
        if i < 0:
            return NAN, NAN, True
        snap_t, (depth, spread) = self._snap_t[i], self._snap_v[i]
        del self._snap_t[:i], self._snap_v[:i]  # keep the as-of snapshot for the next tick
        if ts_ms - snap_t > self.p.stale_ms:
            return NAN, NAN, True
        return depth, spread, False

    def _reference(self, values: deque[float]) -> float:
        arr = np.fromiter(values, float, len(values))
        known = arr[~np.isnan(arr)]
        return float(np.median(known)) if len(known) >= self.min_ref else NAN

    def _tick(self, ts_ms: int) -> OutcomeLabel | None:
        self.stats.ticks += 1
        self.stats.last_tick_ms = ts_ms
        depth, spread, stale = self._state(ts_ms)

        ref_depth, ref_spread = self._reference(self._ref_depth), self._reference(self._ref_spread)
        self._ref_depth.append(depth)  # the reference for later ticks includes this one
        self._ref_spread.append(spread)
        depth_ratio, spread_ratio = depth / ref_depth, spread / ref_spread
        if np.isnan(depth_ratio) or np.isnan(spread_ratio):
            stressed = NAN
        else:
            hit = depth_ratio < self.p.depth_fraction or (
                self.p.use_spread and spread_ratio > self.p.spread_multiple
            )
            stressed = float(hit)
        self._recent_stressed.append(stressed)
        recent = list(self._recent_stressed)
        if len(recent) < self.p.confirm_ticks or any(np.isnan(recent)):
            confirmed = NAN
        else:
            confirmed = min(recent)
        self._history.append(_Tick(ts_ms, stale, stressed, confirmed))
        return self._resolve(ts_ms - self.p.horizon_ticks * self.step)

    def _resolve(self, t: int) -> OutcomeLabel | None:
        """Outcome of tick ``t`` now that tick ``t + H`` is known, or None if ``t`` predates the stream."""
        hist = list(self._history)
        idx = next((i for i, h in enumerate(hist) if h.ts_ms == t), None)
        if idx is None:
            return None
        at, future = hist[idx], hist[idx + 1 :]
        label: int | None
        if at.stressed == 1:
            label, reason = None, "already_stressed"
        elif np.isnan(at.stressed) or any(np.isnan(h.confirmed) for h in future):
            label = None
            window = hist[max(idx - self.p.confirm_ticks + 1, 0) :]
            reason = "gap" if any(h.stale for h in window) else "warmup"
        else:
            label, reason = int(any(h.confirmed == 1 for h in future)), None
        self.stats.outcomes[str(label) if label is not None else reason] += 1
        return OutcomeLabel(
            symbol=self.cfg.symbol, ts_ms=t, horizon_s=self.p.horizon_s, label=label, void_reason=reason
        )


async def run_verifier(
    verifier: OutcomeVerifier, events: AsyncIterator[tuple[str, dict]], publish: Publish
) -> VerifierStats:
    """Consume ``(kind, row)`` events and publish every outcome to ``outcomes`` keyed by symbol."""
    async for kind, row in events:
        for outcome in verifier.on_event(kind, row):
            await publish(OUTCOMES_TOPIC, outcome.symbol, outcome.model_dump(mode="json"))
    return verifier.stats

=== src/sentinalops/training/train.py ===
"""Train any registered model by hand: ``python -m sentinalops.training.train``.

This is the manual upgrade path (``docs/models.md``). The model comes from ``--model`` or the
config's ``model.name``; the data is either synthetic regimes or a training table built by the
offline pipeline (``data/processed/features``).

- **Table data:** only ``label_status == 'ok'`` rows; the newest 20% is held out, after a gap of
  ``split_embargo_seconds`` so no training label looks into the holdout.
- **Synthetic data:** rows are independent draws, so a stratified random split is fine.

The holdout average precision is stamped on the model as ``sentinal_holdout_ap_``; the heal loop
uses it as the reference when this model becomes the champion.
"""

from __future__ import annotations

import argparse
import json
import os
from pathlib import Path

import joblib
import pandas as pd
from sklearn.metrics import average_precision_score
from sklearn.model_selection import train_test_split

from sentinalops.config import DEFAULT_DOMAIN_CONFIG, DomainConfig, load_domain_config
from sentinalops.models import get_recipe
from sentinalops.models.base import check_fitted

DEFAULT_OUT = Path("models/champion.joblib")


def time_split(table: pd.DataFrame, cfg: DomainConfig, holdout_fraction: float = 0.2):
    """(train, holdout): chronological, usable rows only, with the embargo between them."""
    ok = table[table["label_status"] == "ok"].sort_values("ts_ms")
    if ok.empty:
        raise ValueError("no usable rows (label_status == 'ok') in the table")
    cut = int(ok["ts_ms"].iloc[int(len(ok) * (1 - holdout_fraction))])
    train = ok[ok["ts_ms"] < cut - cfg.split_embargo_seconds * 1000]
    return train, ok[ok["ts_ms"] >= cut]


def load_data(data: str, rows: int) -> pd.DataFrame:
    if data == "synthetic":
        from sentinalops.simulator.regimes import make_ticks

        return make_ticks("calm", rows)
    from sentinalops.data.pipeline import read_table

    table = read_table(data)
    if table.empty:
        raise SystemExit(f"no training table under {data}")
    return table


def train(data: pd.DataFrame, cfg: DomainConfig, name: str, params: dict, seed: int = 0):
    """Fit ``name`` and return (model, holdout AP, training rows)."""
    if "ts_ms" in data and "label_status" in data:
        tr, ho = time_split(data, cfg)
    else:
        tr, ho = train_test_split(data, test_size=0.2, stratify=data["label"], random_state=seed)
    model = get_recipe(name).fit(tr, cfg, seed, params)
    check_fitted(model, cfg)
    y = ho["label"].astype(int)
    p = model.predict_proba(ho[cfg.feature_names])[:, 1]
    ap = float(average_precision_score(y, p)) if y.any() else float("nan")
    model.sentinal_holdout_ap_ = ap
    return model, ap, len(tr)


def main(argv: list[str] | None = None) -> None:
    ap = argparse.ArgumentParser(description="Train a registered model and save it as the champion file")
    ap.add_argument("--config", default=os.environ.get("DOMAIN_CONFIG", str(DEFAULT_DOMAIN_CONFIG)))
    ap.add_argument("--model", default=None, help="registered model name (default: config model.name)")
    ap.add_argument("--params", default=None, help="JSON object overriding the recipe's parameters")
    ap.add_argument("--data", default="synthetic", help="'synthetic' or a training-table directory")
    ap.add_argument("--rows", type=int, default=20_000, help="synthetic rows")
    ap.add_argument("--seed", type=int, default=0)
    ap.add_argument("--out", default=os.environ.get("MODEL_PATH", str(DEFAULT_OUT)))
    args = ap.parse_args(argv)

    cfg = load_domain_config(args.config)
    name = args.model or cfg.model.name
    params = json.loads(args.params) if args.params else dict(cfg.model.params)
    model, holdout_ap, n = train(load_data(args.data, args.rows), cfg, name, params, args.seed)

    out = Path(args.out)
    out.parent.mkdir(parents=True, exist_ok=True)
    tmp = out.with_suffix(out.suffix + ".tmp")
    joblib.dump(model, tmp)
    os.replace(tmp, out)
    print(f"model={name} holdout_ap={holdout_ap:.4f} rows={n} out={out}")


if __name__ == "__main__":
    main()

"""Train the baseline stress-forecast champion on calm-regime ticks and save it.

Until recorded data flows through chunk 12, training uses the synthetic regimes in

import joblib
import pandas as pd
from sklearn.ensemble import HistGradientBoostingClassifier

def fit(df: pd.DataFrame, seed: int = 0) -> HistGradientBoostingClassifier:
    """The ``gbm_v1`` recipe on the default features (kept for existing callers)."""
    from types import SimpleNamespace

    from sentinalops.models import get_recipe

    try:
        import mlflow

        with mlflow.start_run(run_name="champion-baseline"):
            mlflow.log_metric("avg_precision", ap)
            mlflow.sklearn.log_model(model, "model")
    except Exception as e:  # MLflow is optional locally

    main()


=== tests/pipeline_helpers.py ===
"""Synthetic recordings in the recorder's on-disk layout, for pipeline tests and the speed check."""

from __future__ import annotations

from pathlib import Path

import numpy as np
import pandas as pd

from sentinalops.data.recorder.messages import AGG_TRADE, DEPTH5
from sentinalops.data.recorder.sink import ParquetSink

H0 = 1_763_200_800_000  # 2025-11-15T10:00:00Z, an hour boundary


def make_events(
    start_ms: int,
    seconds: int,
    *,
    symbol: str = "BTCUSDT",
    snapshot_ms: int = 500,
    trades_per_s: float = 2.0,
    stress: tuple[tuple[int, int], ...] = (),
    holes: tuple[tuple[int, int], ...] = (),
    seed: int = 0,
) -> tuple[pd.DataFrame, pd.DataFrame]:
    """(depth, trades) in the recorder schema.

    ``stress`` is a list of ``(offset_s, duration_s)`` episodes where the book thins to 20%.
    ``holes`` are ``(offset_s, duration_s)`` stretches with no events of either kind.
    """
    rng = np.random.default_rng(seed)
    t = start_ms + np.arange(0, seconds * 1000, snapshot_ms, dtype=np.int64)
    t = t + rng.integers(0, 40, len(t))  # receive jitter
    n = len(t)
    thin = np.ones(n)
    for off, dur in stress:
        thin[(t >= start_ms + off * 1000) & (t < start_ms + (off + dur) * 1000)] = 0.2
    mid = 90_000 + np.cumsum(rng.normal(0, 0.5, n)).round(2)
    depth = {"symbol": symbol, "last_update_id": np.arange(n, dtype=np.int64) + 1_000, "recv_time_ms": t}
    for k in range(1, 6):
        depth[f"bid_px_{k}"] = mid - 0.005 - (k - 1) * 0.01
        depth[f"bid_qty_{k}"] = rng.uniform(1.0, 3.0, n) * thin
        depth[f"ask_px_{k}"] = mid + 0.005 + (k - 1) * 0.01
        depth[f"ask_qty_{k}"] = rng.uniform(1.0, 3.0, n) * thin
    depth = pd.DataFrame(depth)

    m = rng.poisson(trades_per_s * seconds)
    tt = np.sort(start_ms + rng.integers(0, seconds * 1000, m)).astype(np.int64)
    trades = pd.DataFrame(
        {
            "symbol": symbol,
            "event_time_ms": tt - 5,
            "trade_time_ms": tt - 6,
            "agg_trade_id": np.arange(m, dtype=np.int64) + 50_000,
            "price": 90_000.0 + rng.normal(0, 5, m).round(2),
            "quantity": rng.uniform(0.001, 0.2, m),
            "first_trade_id": np.arange(m, dtype=np.int64) * 3,
            "last_trade_id": np.arange(m, dtype=np.int64) * 3 + 2,
            "buyer_is_maker": rng.random(m) < 0.5,
            "recv_time_ms": tt,
        }
    )
    for off, dur in holes:
        lo, hi = start_ms + off * 1000, start_ms + (off + dur) * 1000
        depth = depth[(depth.recv_time_ms < lo) | (depth.recv_time_ms >= hi)]
        trades = trades[(trades.recv_time_ms < lo) | (trades.recv_time_ms >= hi)]
    return depth.reset_index(drop=True), trades.reset_index(drop=True)


def write_recording(root: Path, depth: pd.DataFrame, trades: pd.DataFrame, chunk_s: int = 600) -> None:
    """Write through the real ``ParquetSink``, several files per hour like a live recorder."""
    sink = ParquetSink(root)
    for kind, df in ((DEPTH5, depth), (AGG_TRADE, trades)):
        if len(df) == 0:
            continue
        bucket = df["recv_time_ms"] // (chunk_s * 1000)
        for _, part in df.groupby(bucket, sort=True):
            for row in part.to_dict("records"):
                sink.add(kind, row)
            res = sink.flush()
            assert not res.errors, res.errors

    data = {
        "lastUpdateId": i,
        "bids": [[f"{100 - k * 0.01:.2f}", "1.5"] for k in range(5)],

class FakeExchange:
    """Local WebSocket server. ``script(ws, connection_number)`` drives each connection."""

    def __init__(self, script: Callable[[object, int], Awaitable[None]]) -> None:
        self.script = script
        self.connections = 0
        self.port = 0
        self._server = None
        self._handlers: set[asyncio.Task] = set()

    async def _handler(self, ws, *_):
        self.connections += 1
        self._handlers.add(asyncio.current_task())
        try:
            await self.script(ws, self.connections)
        except websockets.ConnectionClosed:
            pass
        finally:
            self._handlers.discard(asyncio.current_task())

    async def __aenter__(self) -> FakeExchange:
        self._server = await websockets.serve(self._handler, "127.0.0.1", 0)
        self.port = self._server.sockets[0].getsockname()[1]
        return self

    async def __aexit__(self, *exc) -> None:
        for task in list(self._handlers):  # scripts that sleep must not hold up teardown
            task.cancel()
        self._server.close()
        await asyncio.wait_for(self._server.wait_closed(), 10)

    @property
    def base_url(self) -> str:
        return f"ws://127.0.0.1:{self.port}"


def make_recorder(tmp_path, exchange: FakeExchange, **overrides) -> Recorder:
    cfg = dict(
        symbols=("BTCUSDT",),
        data_dir=tmp_path,
        base_url=exchange.base_url,
        allowed_hosts=LOCAL,
        flush_interval_s=3600,
        stale_after_s=5,
        backoff_base_s=0.02,
        backoff_max_s=0.05,
        stats_interval_s=3600,
        open_timeout_s=5,
    )
    cfg.update(overrides)
    return Recorder(RecorderConfig(**cfg))


async def run_until(recorder: Recorder, condition: Callable[[], bool], timeout: float = 15.0) -> None:
    """Run the recorder until ``condition()`` is true, then stop it gracefully."""
    task = asyncio.create_task(recorder.run())
    deadline = time.monotonic() + timeout
    try:
        while not condition():
            if task.done():
                task.result()
                raise AssertionError("recorder exited before the condition was met")
            if time.monotonic() > deadline:
                raise AssertionError(f"timed out; stats={recorder.stats}")
            await asyncio.sleep(0.01)
    finally:
        recorder.request_stop()
        await asyncio.wait_for(task, 10)

def make_depth(
    seconds: int, depth=lambda s: 100.0, spread_ticks=lambda s: 1, skip=lambda s: False
) -> pd.DataFrame:
    """Snapshots every 100 ms. ``depth(s)`` is total top-5 quantity at second-offset ``s`` (float)."""
    rows = []
    for k in range(seconds * 10):
        s = k / 10
        if skip(s):
            continue
        bid, ask = 100.00, 100.00 + 0.01 * spread_ticks(s)
        row = {"symbol": "BTCUSDT", "last_update_id": k, "recv_time_ms": T0 + k * 100}
        for i in range(1, 6):
            row[f"bid_px_{i}"], row[f"bid_qty_{i}"] = bid - 0.01 * (i - 1), depth(s) / 10
            row[f"ask_px_{i}"], row[f"ask_qty_{i}"] = ask + 0.01 * (i - 1), depth(s) / 10
        rows.append(row)
    return pd.DataFrame(rows)


def at(labeled: pd.DataFrame, second: int) -> pd.Series:
    return labeled.loc[T0 + second * 1000]


def random_depth(seconds: int, seed: int = 7) -> pd.DataFrame:
    """A noisy book with sudden liquidity withdrawals and a couple of feed gaps."""
    rng = np.random.default_rng(seed)
    level = 100 * np.exp(np.cumsum(rng.normal(0, 0.01, seconds)))
    crash = np.zeros(seconds, dtype=bool)
    for start in rng.choice(np.arange(60, seconds - 20), size=seconds // 60, replace=False):
        crash[start : start + rng.integers(1, 12)] = True
    level = np.where(crash, level * 0.3, level)
    gaps = [(200, 209), (400, 404)] if seconds > 420 else []
    return make_depth(
        seconds, depth=lambda s: float(level[int(s)]), skip=lambda s: any(a <= s < b for a, b in gaps)
    )

import asyncio
import os

import pytest

from sentinalops.live.bus import InMemoryBus, stable_prediction_id

BUSES = ["memory"] + (["kafka"] if os.environ.get("KAFKA_BROKER") else [])


def make_bus(kind: str, **kw):
    if kind == "memory":
        return InMemoryBus(**kw)
    from sentinalops.live.kafka_bus import KafkaBus

    return KafkaBus(os.environ["KAFKA_BROKER"], client_id=f"test-{os.getpid()}")


async def take(sub, n, timeout=10.0):
    out = []
    async with asyncio.timeout(timeout):
        async for msg in sub:
            out.append(msg)
            if len(out) == n:
                break
    return out


@pytest.mark.parametrize("kind", BUSES)
def test_bus_fanout_and_order(kind):
    async def main():
        bus = make_bus(kind)
        a = await bus.subscribe(["raw_depth", "raw_trades"], group="a")
        b = await bus.subscribe(["raw_depth"], group="b")
        for i in range(5):
            await bus.publish("raw_depth", "BTCUSDT", {"i": i})
        await bus.publish("raw_trades", "BTCUSDT", {"i": 99})
        got_a, got_b = await take(a, 6), await take(b, 5)
        await bus.close()
        return got_a, got_b

    got_a, got_b = asyncio.run(main())
    assert [v["i"] for t, v in got_a if t == "raw_depth"] == [0, 1, 2, 3, 4]
    assert ("raw_trades", {"i": 99}) in got_a
    assert [v["i"] for _, v in got_b] == [0, 1, 2, 3, 4]

    async def main():

def test_heal_fires_from_ledger_data(tmp_path, calm_model):
    """Calm then volatile, scored into the ledger with features; the service retrains from it."""
    table = S.market([("calm", 90), ("volatile", 150)], T0)
    names, h = CFG.feature_names, CFG.label_horizon_seconds * 1000
    reg, led = ModelRegistry(tmp_path / "models"), Ledger(tmp_path / "l.sqlite")
    joblib.dump(calm_model, reg.champion_path)
    svc = HealService(CFG, led, reg)
    served = {"champion": ChampionModel(reg.champion_path, names),
              "shadow": ChampionModel(reg.shadow_path, names)}
    actions, matured_until = [], T0
    for lo in range(T0, T0 + 240 * MIN, 5 * MIN):
        hi = lo + 5 * MIN
        chunk = table[(table["ts_ms"] >= lo) & (table["ts_ms"] < hi)]
        preds = []
        for model_ref in served.values():
            if not model_ref.available:
                continue
            model, version = model_ref.current()
            probs = model.predict_proba(chunk[names])[:, 1]
            for (_, row), p in zip(chunk.iterrows(), probs, strict=True):
                preds.append((StressPrediction(symbol="BTCUSDT", ts_ms=int(row.ts_ms), horizon_s=30,
                                               probability=float(p), alert=bool(p >= 0.5), threshold=0.5,
                                               model_version=version),
                              {n: float(row[n]) for n in names}))
        matured = table[(table["ts_ms"] >= matured_until) & (table["ts_ms"] + h < hi)]
        led.add_many(preds, [OutcomeLabel(symbol="BTCUSDT", ts_ms=int(t), horizon_s=30, label=int(y))
                             for t, y in zip(matured["ts_ms"], matured["label"], strict=True)])
        matured_until = hi - h
        actions.append(((hi - T0) // MIN, svc.tick(hi).action))

    assert all(a == "idle" for m, a in actions if m <= 90), actions
    assert any(a.startswith("retrained") for m, a in actions if m > 90), actions
    assert reg.count("promote") >= 2  # the bootstrap, then a healed model

@pytest.fixture(scope="module")

    return make_events(H0, 1800, stress=((900, 40),), holes=((1300, 10),), seed=21)

    path = tmp_path_factory.mktemp("m") / "champion.joblib"
    joblib.dump(fit(make_ticks("calm", 4000, seed=2)), path)
    return path


def arrival_order(depth, trades):
    order = merge_events(trades, depth)
    rows = {DEPTH5: depth.to_dict("records"), AGG_TRADE: trades.to_dict("records")}
    return [(k, rows[k][i]) for k, i in zip(order["kind"], order["row"], strict=True)]


async def replay(seq, ledger, model_path, seconds=None, shadow_path=None):
    """Deterministic event time: publish one second of events, drain, then step the clock."""
    bus = InMemoryBus()
    proc = StreamProcessor(CFG, model=ChampionModel(model_path, NAMES),
                           shadow=ChampionModel(shadow_path, NAMES) if shadow_path else None)
    verifier = OutcomeVerifier(CFG)
    tasks, stats = await start_pipeline(bus, proc, verifier, ledger)
    by_second: dict[int, list] = {}
    for kind, row in seq:
        by_second.setdefault(row["recv_time_ms"] // 1000, []).append((kind, row))
    first, last = min(by_second), max(by_second)
    end = first + seconds if seconds else last + 1
    for s in range(first, end):
        for kind, row in by_second.get(s, []):
            await bus.publish(TOPIC_OF_KIND[kind], row["symbol"], row)
        await bus.drain()
        await clock_step(bus, proc, verifier, (s + 1) * 1000, stats)
        await bus.drain()
    await bus.close()
    await asyncio.gather(*tasks)
    return proc, verifier, stats


def test_replay_matches_offline(events, model_file, tmp_path):
    depth, trades = events
    led = Ledger(tmp_path / "l.sqlite")
    proc, verifier, stats = asyncio.run(replay(arrival_order(depth, trades), led, model_file))
    rows = led.predictions(0)
    assert len(rows) == proc.stats.predictions > 1000 and stats.ledger_errors == 0

    offline = compute_ticks(depth, trades, CFG.feature_params())
    offline = offline[offline["valid"]]
    online = pd.DataFrame([{**r["features"], "ts_ms": r["ts_ms"]} for r in rows])
    online = online.set_index("ts_ms").sort_index()
    assert set(online.index) <= set(offline.index)
    np.testing.assert_allclose(online[NAMES].to_numpy(), offline.loc[online.index, NAMES].to_numpy(),
                               rtol=1e-9, atol=1e-12)

    labels = build_labels(depth, CFG.label_params())["label"]
    matured = [r for r in rows if r["matured"]]
    assert len(matured) > 1000
    for r in matured:
        want = labels.get(r["ts_ms"], math.nan)
        got = math.nan if r["label"] is None else float(r["label"])
        assert (math.isnan(want) and math.isnan(got)) or want == got, r["ts_ms"]
    assert any(r["void_reason"] == "gap" for r in matured)
    assert proc.stats.late == {AGG_TRADE: 0, DEPTH5: 0} and verifier.stats.late == 0


def test_silent_feed_suspends_ticks(events, model_file, tmp_path):
    depth, trades = events
    cut = H0 + 120_000
    seq = arrival_order(depth[depth.recv_time_ms < cut], trades[trades.recv_time_ms < cut])

    async def main():
        bus = InMemoryBus()
        proc = StreamProcessor(CFG, model=ChampionModel(model_file, NAMES))
        verifier = OutcomeVerifier(CFG)
        tasks, stats = await start_pipeline(bus, proc, verifier, Ledger(tmp_path / "l.sqlite"))
        for kind, row in seq:
            await bus.publish(TOPIC_OF_KIND[kind], row["symbol"], row)
        await bus.drain()
        before = proc.stats.predictions
        # The feed goes silent; only the wall clock moves. The book stays fresh for
        # stale_data_seconds (3 s), so a few more ticks are scored, then scoring must stop.
        for s in range(1, 6):
            await clock_step(bus, proc, verifier, cut + s * 1000, stats)
        after_5s, suspended_5s = proc.stats.predictions, proc.stats.suspended_ticks
        for s in range(6, 16):
            await clock_step(bus, proc, verifier, cut + s * 1000, stats)
        await bus.drain()
        await bus.close()
        await asyncio.gather(*tasks)
        return proc, before, after_5s, suspended_5s

    proc, before, after_5s, suspended_5s = asyncio.run(main())
    assert before > 30 and after_5s - before <= CFG.stale_data_seconds + 1
    assert proc.stats.predictions == after_5s, "no predictions once the book is stale"
    assert proc.stats.suspended_ticks - suspended_5s == 10

def snapshot(t_ms, bid=100.00, ask=100.01, bid_qty=1.0, ask_qty=1.0, i=0):
    row = {"symbol": "BTCUSDT", "last_update_id": i, "recv_time_ms": int(t_ms)}
    for k in range(1, 6):
        row[f"bid_px_{k}"], row[f"bid_qty_{k}"] = bid - 0.01 * (k - 1), bid_qty
        row[f"ask_px_{k}"], row[f"ask_qty_{k}"] = ask + 0.01 * (k - 1), ask_qty
    return row


def book(seconds, step_ms=100, **fn):
    """Snapshots every ``step_ms``; each keyword is a function of the second offset."""
    rows = []
    for k in range(int(seconds * 1000 / step_ms) + 1):
        s = k * step_ms / 1000
        rows.append(snapshot(T0 + k * step_ms, i=k, **{name: f(s) for name, f in fn.items()}))
    return pd.DataFrame(rows)


def trade_rows(items):
    """items: (second offset, quantity, buyer_is_maker)."""
    return pd.DataFrame(
        [{"recv_time_ms": int(T0 + s * 1000), "quantity": q, "buyer_is_maker": m} for s, q, m in items],
        columns=["recv_time_ms", "quantity", "buyer_is_maker"],
    )


NO_TRADES = trade_rows([])


def random_market(seconds, seed=11, gaps=((200, 206), (410, 412))):
    """Jittered snapshots, a wandering price, random sizes, bursty trades and two feed gaps."""
    rng = np.random.default_rng(seed)
    times = np.cumsum(rng.integers(60, 140, size=seconds * 10)) + T0
    times = times[times <= T0 + seconds * 1000]
    keep = np.ones(len(times), dtype=bool)
    for a, b in gaps:
        keep &= ~((times >= T0 + a * 1000) & (times < T0 + b * 1000))
    times = times[keep]
    mid = 100 * np.exp(np.cumsum(rng.normal(0, 2e-5, len(times))))
    rows = []
    for i, (t, m) in enumerate(zip(times, mid, strict=True)):
        half = 0.005 * rng.integers(1, 4)
        row = snapshot(t, bid=m - half, ask=m + half, i=i)
        for k in range(1, 6):
            row[f"bid_qty_{k}"], row[f"ask_qty_{k}"] = rng.uniform(0.01, 3), rng.uniform(0.01, 3)
        rows.append(row)
    n = seconds * 6
    tt = np.sort(rng.integers(T0, T0 + seconds * 1000, size=n))
    tt = tt[np.isin(tt // 1000, [(T0 // 1000) + s for a, b in gaps for s in range(a, b)], invert=True)]
    trades = pd.DataFrame(
        {
            "recv_time_ms": tt,
            "quantity": rng.exponential(0.05, len(tt)),
            "buyer_is_maker": rng.random(len(tt)) < 0.5,
        }
    )
    return pd.DataFrame(rows), trades

def test_termination_signal_flushes_buffered_rows(tmp_path):
    """SIGTERM on POSIX; Ctrl-Break on Windows, where SIGTERM cannot be caught by another process."""
    out = tmp_path / "data"
    code = textwrap.dedent(CHILD).format(tests=str(Path(__file__).parent), out=str(out))
    env = {**os.environ, "PYTHONPATH": str(Path(__file__).parent.parent / "src")}
    flags = subprocess.CREATE_NEW_PROCESS_GROUP if sys.platform == "win32" else 0
    proc = subprocess.Popen(
        [sys.executable, "-c", code],
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
        env=env,
        creationflags=flags,
    )
    try:
        assert proc.stdout.readline().strip() == "READY", proc.stderr.read()
        assert not list(out.rglob("part-*.parquet"))  # nothing on disk yet
        time.sleep(0.1)
        proc.send_signal(signal.CTRL_BREAK_EVENT if sys.platform == "win32" else signal.SIGTERM)
        stdout, stderr = proc.communicate(timeout=20)
    finally:
        if proc.poll() is None:
            proc.kill()
    assert proc.returncode == 0, stderr
    received = int(stdout.split("STOPPED")[1])
    ids = _column(out, DEPTH5, "last_update_id")
    assert len(ids) == received >= 50 and ids == list(range(1, received + 1))
    assert not list(out.rglob("*.tmp"))

def test_rotation_with_a_lagging_old_connection_loses_nothing(tmp_path):
    """The old link is 200 ms behind when the new one connects: its backlog must still be recorded."""
    t0 = time.monotonic()

    def index_at(delay: float) -> int:
        return int((time.monotonic() - t0 - delay) / 0.005)

    async def script(ws, n):
        delay = 0.2 if n == 1 else 0.0  # the first connection is congested
        sent = index_at(delay) if n > 1 else 0
        while True:
            latest = index_at(delay)
            for i in range(sent + 1, latest + 1):
                await ws.send(trade_frame(i))
                await ws.send(depth_frame(i))
            sent = max(sent, latest)
            await asyncio.sleep(0.005)

    async def go():
        async with FakeExchange(script) as ex:
            rec = make_recorder(tmp_path, ex, max_connection_s=0.6)
            await run_until(rec, lambda: rec.stats.rotations >= 1 and rec.stats.messages[DEPTH5] > 250)
            return rec

    rec = asyncio.run(go())
    assert rec.stats.rotations >= 1 and rec.stats.reconnects == 0
    for kind, column in ((AGG_TRADE, "agg_trade_id"), (DEPTH5, "last_update_id")):
        ids = _column(tmp_path, kind, column)
        missing = sorted(set(range(ids[0], ids[-1] + 1)) - set(ids))
        assert not missing, f"{kind}: {len(missing)} ids lost across the rotation, e.g. {missing[:5]}"


def test_rotation_records_a_gap_when_the_old_connection_never_catches_up(tmp_path):
    t0 = time.monotonic()
    second_open = asyncio.Event()

    def index_now() -> int:
        return int((time.monotonic() - t0) / 0.005)

    async def script(ws, n):
        if n > 1:
            second_open.set()
        delay = 0.3 if n == 1 else 0.0  # far behind, and goes silent once replaced
        sent = index_now() if n > 1 else 0
        while not (n == 1 and second_open.is_set()):
            latest = int((time.monotonic() - t0 - delay) / 0.005)
            for i in range(sent + 1, latest + 1):
                await ws.send(trade_frame(i))
                await ws.send(depth_frame(i))
            sent = max(sent, latest)
            await asyncio.sleep(0.005)
        await asyncio.sleep(30)

    async def go():
        async with FakeExchange(script) as ex:
            rec = make_recorder(tmp_path, ex, max_connection_s=0.6, rotation_drain_s=0.3, stale_after_s=5)

            return rec

    rec = asyncio.run(go())
    assert rec.stats.rotation_drain_timeouts == 1 and rec.stats.reconnects == 0
    report = audit_recording.audit(tmp_path)

def replay(depth, trades, delay_ms=None):
    """``(kind, row)`` in arrival order. ``delay_ms`` per kind simulates two topics arriving skewed."""
    order = merge_events(trades, depth)
    rows = {DEPTH5: depth.to_dict("records"), AGG_TRADE: trades.to_dict("records")}
    seq = [(k, rows[k][i]) for k, i in zip(order["kind"], order["row"], strict=True)]
    if delay_ms:
        seq.sort(key=lambda e: e[1]["recv_time_ms"] + delay_ms.get(e[0], 0))  # stable: ties keep order
    return seq


def until(events, ms):
    depth, trades = events
    return depth[depth.recv_time_ms < H0 + ms], trades[trades.recv_time_ms < H0 + ms]


def feed(proc, seq):
    out = []
    for kind, row in seq:
        out += proc.on_event(kind, row)
    return out


def ticks_of(out):
    return {m.ts_ms: m.features for topic, m in out if topic == FEATURES_TOPIC}

"""The project thesis on synthetic regimes (task 3.5).

A champion trained on the calm regime degrades when the market turns volatile, because stress
there has different precursors; a challenger retrained on recent volatile data recovers.
"""

import numpy as np
import pandas as pd
import pytest

from sentinalops.features.microstructure import FEATURE_ORDER
from sentinalops.simulator.regimes import BASE_RATE, make_ticks
from sentinalops.training.train_baseline import evaluate, fit, train


@pytest.fixture(scope="module")
def data():
    return {
        "calm_train": make_ticks("calm", 20_000, seed=0),
        "calm_test": make_ticks("calm", 20_000, seed=1),
        "vol_train": make_ticks("volatile", 20_000, seed=2),
        "vol_test": make_ticks("volatile", 20_000, seed=3),
    }


def test_generator_shape_and_determinism():
    a, b = make_ticks("calm", 500, seed=9), make_ticks("calm", 500, seed=9)
    assert list(a.columns) == [*FEATURE_ORDER, "label"]
    pd.testing.assert_frame_equal(a, b)
    assert np.isfinite(a[FEATURE_ORDER].to_numpy()).all()
    assert (a["depth_top5"] > 0).all() and (a["rel_spread"] > 0).all()
    np.testing.assert_allclose(a["bid_depth"] + a["ask_depth"], a["depth_top5"])
    with pytest.raises(ValueError):
        make_ticks("sideways")


def test_regimes_differ_in_base_rate_and_precursors(data):
    calm, vol = data["calm_train"], data["vol_train"]
    assert calm["label"].mean() == pytest.approx(BASE_RATE["calm"], abs=0.005)
    assert vol["label"].mean() == pytest.approx(BASE_RATE["volatile"], abs=0.008)

    def gap(df, col):  # how much a feature moves before stress, in noise units
        return (df.loc[df.label == 1, col].mean() - df.loc[df.label == 0, col].mean()) / df[col].std()

    # Calm: the book thins out before stress. Volatile: trading bursts instead.
    assert gap(calm, "depth_change_60s") < -0.5 and abs(gap(vol, "depth_change_60s")) < 0.2
    assert gap(vol, "trade_intensity_1s") > 0.5 and abs(gap(calm, "trade_intensity_1s")) < 0.1


def test_champion_degrades_then_challenger_recovers(data):
    champion, calm_ap = train(data["calm_train"])
    vol_ap = evaluate(champion, data["vol_test"])
    assert calm_ap > 0.4, "champion should learn the calm regime"
    assert vol_ap < 0.2 and vol_ap < calm_ap / 2, "champion should fail when the regime shifts"

    recent = pd.concat([data["calm_train"].sample(5_000, random_state=0), data["vol_train"]])
    challenger = fit(recent)
    healed_ap = evaluate(challenger, data["vol_test"])
    assert healed_ap > 0.5 and healed_ap > vol_ap + 0.3, "retraining on recent data should recover"
    assert evaluate(challenger, data["calm_test"]) > 0.3, "and not forget the calm regime entirely"

def test_ledger_add_many_is_one_idempotent_batch(tmp_path):
    led = Ledger(tmp_path / "l.sqlite")
    t0 = 1_763_200_800_000
    preds = [(StressPrediction(prediction_id=f"{i:032x}", symbol="BTCUSDT", ts_ms=t0 + i * 1000,
                               horizon_s=30, probability=0.2, alert=False, threshold=0.5,
                               model_version="v1"), {"a": float(i)})
             for i in range(3)]
    outs = [OutcomeLabel(symbol="BTCUSDT", ts_ms=t0 + i * 1000, horizon_s=30, label=i % 2) for i in range(2)]

    assert len(rows) == 3 and rows[0]["features"] == {"a": 0.0} and rows[1]["label"] == 1
